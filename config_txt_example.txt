# ============================================================================
# RPi5 HAOS config.txt for Waveshare Isolated Industrial Interface Board
# Board: RS232/RS485/CAN/CAN-FD Expansion Board
# Location on HAOS: boot partition (mmcblk0p1) /config.txt
# ============================================================================
#
# VERIFIED HARDWARE:
#   - 2x RS485: SC16IS752 + SP485 via SPI1 -> /dev/ttySC0, /dev/ttySC1
#   - 1x RS232: SP3232EEN via UART0       -> /dev/ttyAMA0
#   - 1x CAN:   MCP2515 + SN65HVD230 via SPI0 CE0 -> can1
#   - 1x CAN FD: MCP2518FD + MCP2562FD via SPI0 CE1 -> can0
#
# GPIO Pin Mapping:
#   SC16IS752 INT = GPIO 25 (Pin 22)
#   MCP2515 INT   = GPIO 23 (Pin 16)
#   MCP2518FD INT = GPIO 24 (Pin 18)
#   UART0 TX/RX   = GPIO 14/15 (Pin 8/10) - RS232
#   SPI0 CE0       = GPIO 8 (Pin 24) - MCP2515
#   SPI0 CE1       = GPIO 7 (Pin 26) - MCP2518FD
#   SPI1 CE0       = GPIO 18 (Pin 12) - SC16IS752
#
# ============================================================================

# --- Add these lines to config.txt (in the [all] section or before [cm4]) ---

# Enable SPI bus
dtparam=spi=on

# RS485 x2: SC16IS752 dual UART via SPI1
dtoverlay=spi1-3cs
dtoverlay=sc16is752-spi1,int_pin=25

# CAN Classic: MCP2515 on SPI0 CE0, 16MHz oscillator, INT=GPIO23
dtoverlay=mcp2515,spi0-0,oscillator=16000000,interrupt=23

# CAN FD: MCP2518FD on SPI0 CE1, INT=GPIO24
dtoverlay=mcp251xfd,spi0-1,interrupt=24

# RS232: Enable UART0 on RPi5 (GPIO 14/15)
dtoverlay=uart0-pi5
