# ============================================================================
# RPi5 HAOS config.txt for RS485 x2, RS232 x1, CAN bus x2
# Location on HAOS: /mnt/boot/config.txt
# ============================================================================
#
# HARDWARE ASSUMPTION:
#   - 2x RS485 transceivers (e.g., SP3485/MAX485) connected to UART1 and UART2
#   - 1x RS232 transceiver (e.g., MAX3232) connected to UART3
#   - 2x CAN bus controllers (MCP2515) connected via SPI0 CE0 and SPI0 CE1
#
# RPi5 UART mapping:
#   UART0 = /dev/ttyAMA0 (GPIO 14/15) - reserved for console/bluetooth
#   UART1 = /dev/ttyAMA1 (GPIO 0/1)   - RS485 #1
#   UART2 = /dev/ttyAMA2 (GPIO 4/5)   - RS485 #2
#   UART3 = /dev/ttyAMA3 (GPIO 8/9)   - RS232
#   UART4 = /dev/ttyAMA4 (GPIO 12/13) - available
#
# CAN bus mapping:
#   SPI0 CE0 = can0 (INT on GPIO 25)
#   SPI0 CE1 = can1 (INT on GPIO 24)
#
# ============================================================================

[all]
# --- Enable UART ---
enable_uart=1

# --- Disable Bluetooth to free UART0 (optional, if using UART0) ---
# dtoverlay=disable-bt

# --- RS485 #1: Enable UART1 on GPIO 0/1 ---
dtoverlay=uart1-pi5

# --- RS485 #2: Enable UART2 on GPIO 4/5 ---
dtoverlay=uart2-pi5

# --- RS232: Enable UART3 on GPIO 8/9 ---
dtoverlay=uart3-pi5

# --- CAN Bus: Enable SPI ---
dtparam=spi=on

# --- CAN #1: MCP2515 on SPI0 CE0, INT=GPIO25, 8MHz/12MHz/16MHz crystal ---
# Adjust oscillator value to match your MCP2515 crystal frequency:
#   8000000  = 8MHz
#   12000000 = 12MHz (common on Waveshare HATs)
#   16000000 = 16MHz
dtoverlay=mcp2515-can0,oscillator=12000000,interrupt=25,spimaxfrequency=2000000

# --- CAN #2: MCP2515 on SPI0 CE1, INT=GPIO24 ---
dtoverlay=mcp2515-can1,oscillator=12000000,interrupt=24,spimaxfrequency=2000000
